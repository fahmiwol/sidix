```python
def kalkulator():
    print("Pilihan Operasi:")
    print("1. Penjumlahan")
    print("2. Pengurangan")
    print("3. Perkalian")
    print("4. Pembagian")

    # Meminta masukan dari pengguna untuk pilih operasi
    pilihan = input("Masukkan nomor operasi (1/2/3/4): ")

    # Memeriksa validitas input
    if pilihan not in ['1', '2', '3', '4']:
        print("Pilihan tidak valid. Silakan masukkan 1, 2, 3, atau 4.")
        return

    # Meminta dua angka dari pengguna
    num1 = float(input("Masukkan bilangan pertama: "))
    num2 = float(input("Masukkan bilangan kedua: "))

    if pilihan == '1':
        hasil = num1 + num2
        print(f"Hasil penjumlahan {num1} + {num2} adalah {hasil}")
    elif pilihan == '2':
        hasil = num1 - num2
        print(f"Hasil pengurangan {num1} - {num2} adalah {hasil}")
    elif pilihan == '3':
        hasil = num1 * num2
        print(f"Hasil perkalian {num1} * {num2} adalah {hasil}")
    elif pilihan == '4':
        if num2 != 0:
            hasil = num1 / num2
            print(f"Hasil pembagian {num1} / {num2} adalah {hasil}")
        else:
            print("Pembagian oleh nol tidak diperbolehkan.")

# Menjalankan fungsi kalkulator
kalkulator()
```
